cmake_minimum_required(VERSION 3.20)
project(FORTIS LANGUAGES C CXX CUDA)

set(MLIR_DIR "" CACHE PATH "<llvm build>/lib/cmake/mlir of the flang+torch-mlir toolchain")
set(LLVM_DIR "" CACHE PATH "<llvm build>/lib/cmake/llvm")
set(CUDNN_DIR "" CACHE PATH "cuDNN root with include/ and lib/")
set(MATH_LIBS_DIR "" CACHE PATH "directory holding lib64/libcublas.so and lib64/libcufft.so")
set(CMAKE_CUDA_ARCHITECTURES 70 CACHE STRING "GPU architecture")

find_package(MLIR REQUIRED CONFIG)
find_package(LLVM REQUIRED CONFIG)
list(APPEND CMAKE_MODULE_PATH "${MLIR_CMAKE_DIR}" "${LLVM_CMAKE_DIR}")
include(TableGen)
include(AddLLVM)
include(AddMLIR)
include(HandleLLVMOptions)
include_directories(${LLVM_INCLUDE_DIRS} ${MLIR_INCLUDE_DIRS})
add_definitions(${LLVM_DEFINITIONS})
find_package(CUDAToolkit REQUIRED)

add_subdirectory(compiler)
add_subdirectory(runtime)

configure_file(driver/fortisc.in ${CMAKE_BINARY_DIR}/fortisc @ONLY)
install(PROGRAMS ${CMAKE_BINARY_DIR}/fortisc driver/fortis_outline.py driver/fortis_hostinfo.py driver/fortis_hostlift.py frontend/tf2linalg.sh DESTINATION bin)
